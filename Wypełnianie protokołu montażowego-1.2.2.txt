(function() {
    'use strict';

    const _d = s => atob(s);
    const _n = s => (s || "").normalize("NFD").replace(/[\u0300-\u036f]/g, "").toUpperCase();
    const _m = (co, ...enc) => enc.some(e => _n(co).includes(_d(e)));

    // --- BEZPIECZNE SPRAWDZANIE NAGŁÓWKA ---
    const headerCaption = document.getElementById('bottom_header').children[1].children[0].innerText;
    if(headerCaption.includes("Protokół montażowy") && document.getElementById("take-trigger")) {

        const myTextbox = document.createElement("div");
        myTextbox.innerHTML = `
            <div style="width:100%">
                <input
                    type="text"
                    id="myTextbox"
                    style="width:100%"
                >
                <button
                    style="border:1px solid #28bea9;cursor:pointer;"
                    id="newButton"
                    onmouseover="this.style.opacity=0.8"
                    onmouseout="this.style.opacity=1">
                    Wypełnij protokół
                </button>
            </div>`;

        document.getElementById("header").appendChild(myTextbox);
        document.getElementById("newButton").addEventListener('click', fillProtocol);

        myTextbox.addEventListener('input', e => {
            if(e.target.value.startsWith('"')) e.target.value = e.target.value.replace(/""/g, '"').slice(1, -1);
        });

        // ==========================================
        // GŁÓWNA FUNKCJA ORKIESTRUJĄCA
        // ==========================================
        async function fillProtocol() {
            $loader.show();
            try {
                const userJSON = JSON.parse(document.getElementById("myTextbox").value);

                if (userJSON.monter && userJSON.monter.includes(' - ')) {
                    userJSON.monter = userJSON.monter.split(' - ').pop().trim();
                }

                const protocolType = userJSON.typ || "";

                await handleCompanyAndType(userJSON, protocolType);
                handleVehicle(userJSON, protocolType);
                await handleRejestrator(userJSON);
                handleCAN(userJSON);
                handleD8AndTacho(userJSON);
                await handleDINs(userJSON);
                await handleTablet(userJSON);
                await handleOtherDevices(userJSON);
                await handleSondy(userJSON);
                handleFinalInfo(userJSON);

                // --- SIM na sam koniec, by nie przerywać niczego wcześniej ---
                if (userJSON.sim) {
                    const pType = protocolType.trim().toLowerCase();
                    const setSim = () => {
                        const $simInput = $('input[name="nr_karty_sim"]');
                        if($simInput.length) {
                            $simInput.val(userJSON.sim).trigger('input').trigger('change').removeClass('mValidatorError');
                        }
                    };

                    setSim();

                    if (pType.includes("upgrade") || pType.includes("przekładka") || pType.includes("przekladka")) {
                        setTimeout(setSim, 1000);
                        setTimeout(setSim, 2500);
                        setTimeout(setSim, 4500);
                    }
                }

                // --- Wybór pojazdu dla upgrade/przekładki NA SAMYM KOŃCU ---
                await handleVehicleSelection(userJSON, protocolType);

            } catch (error) {
                alert("Błąd: " + error.message);
                console.error(error);
            } finally {
                $loader.hide();
            }
        }

        // ==========================================
        // MODUŁY LOGICZNE (KLOCKI)
        // ==========================================

        // ==========================================
        // POMOCNIK: czy pole select2 jest puste/niewybrane?
        // ==========================================

        function isSelect2Empty(selector) {
            try {
                const data = $(selector).select2('data');
                if (!data) return true;
                const text = (data.text || data.id || "").trim();
                return text === "" || text === "-- Wybierz --" || text.toLowerCase().includes("wybierz");
            } catch(e) {
                return true;
            }
        }

        async function handleCompanyAndType(userJSON, protocolType) {

            const pType = protocolType.trim().toLowerCase();

            if (pType !== "") {
                // Kategoria – ustaw tylko jeśli pusta
                const katSelector = $('#kategoria_id').length ? '#kategoria_id' : '#s2id_kategoria_id';
                if (isSelect2Empty(katSelector)) {
                    const katVal = (userJSON.monter === "Monter klienta" || !userJSON.monter) ? 2 : 1;
                    $(katSelector).select2('val', katVal).trigger('change');
                }

                // Typ montażu – ustaw tylko jeśli pusty
                if (isSelect2Empty('#type_id')) {
                    if(pType.includes("montaż") || pType.includes("montaz")) {
                        isInvoiceActiveOnAnotherVehicle(userJSON.id);
                        $("#type_id").select2('val', 1).trigger('change');
                    } else if(pType.includes("upgrade")) {
                        $("#type_id").select2('val', 2).trigger('change');
                    } else if(pType.includes("przekładka") || pType.includes("przekladka")) {
                        $("#type_id").select2('val', 3).trigger('change');
                    }
                }

                // Oddaj wątek przeglądarce żeby zdążyła wykonać synchroniczne handlery change
                await new Promise(r => setTimeout(r, 0));
            }

            // Czytamy AKTUALNIE wybrany typ z protokołu (po ewentualnym ustawieniu powyżej)
            const actualTypeId = parseInt($("#type_id").select2('val')) || 0;

            // Firma – zachowaj istniejącą logikę (już sprawdza czy pusta)
            const firmaWProtokole = $('#firma1_id').select2('data') ? $('#firma1_id').select2('data').text : "";
            if (firmaWProtokole === "" || firmaWProtokole === "-- Wybierz --" || firmaWProtokole.includes("Wybierz")) {
                if(!setSelect2ByText('#firma1_id', userJSON.firma)) alert(`Nie znaleziono firmy ${userJSON.firma}`);
            } else if(firmaWProtokole !== userJSON.firma) {
                alert(`Firma w protokole "${firmaWProtokole}" różni się od podanej: "${userJSON.firma}"`);
            }

            // Czekaj aż grupa pojazdów zostanie odblokowana po zmianie firmy (tylko jeśli jest disabled)
            if ($('#s2id_grupa_pojazdow_id').hasClass('select2-container-disabled')) {
                for (let i = 0; i < 60; i++) {
                    if (!$('#s2id_grupa_pojazdow_id').hasClass('select2-container-disabled')) break;
                    await new Promise(r => setTimeout(r, 50));
                }
            }

            // Grupa pojazdów – ustaw tylko jeśli pusta, opcje są zawsze w DOM
            if (isSelect2Empty('#grupa_pojazdow_id')) {
                if (!setSelect2ByText('#grupa_pojazdow_id', 'Wszystkie', true)) {
                    setSelect2ByText('#grupa_pojazdow_id', 'Wszystkie', false);
                }
            }
        }

        // Wywoływane NA KOŃCU – po załadowaniu całego protokołu, gdy listy są gotowe
        async function handleVehicleSelection(userJSON, protocolType) {
            const actualTypeId = parseInt($("#type_id").select2('val')) || 0;
            const actualIsUpgrade    = actualTypeId === 2;
            const actualIsPrzekladka = actualTypeId === 3;

            if (actualIsPrzekladka) {
                let rejPrzekladka = "";
                if (userJSON.przekladkaZ && userJSON.przekladkaZ.trim() !== "") {
                    rejPrzekladka = userJSON.przekladkaZ.trim();
                } else if (protocolType.includes("Przekładka z ")) {
                    rejPrzekladka = protocolType.split("Przekładka z ")[1].trim();
                }

                if (rejPrzekladka && isSelect2Empty('#old_reg_number')) {
                    if (!setSelect2ByText('#old_reg_number', rejPrzekladka, true)) {
                        setSelect2ByText('#old_reg_number', rejPrzekladka, false);
                    }
                }
            }

            if (actualIsUpgrade) {
                // Czekaj aż pole będzie widoczne i dostępne (max 5s)
                for (let i = 0; i < 100; i++) {
                    const $s2 = $('#s2id_reg_number_select');
                    const isVisible = $s2.is(':visible');
                    const isEnabled = !$s2.hasClass('select2-container-disabled');
                    if (isVisible && isEnabled) break;
                    await new Promise(r => setTimeout(r, 50));
                }

                // Czekaj aż wszystkie AJAXe (wywołane przez zmianę firmy/grupy) się skończą
                // Pole rejestracji jest resetowane przez te requesty - musimy poczekać aż ucichną
                let quietTime = 0;
                for (let i = 0; i < 200; i++) { // max 10s
                    await new Promise(r => setTimeout(r, 50));
                    if (typeof $.active !== 'undefined' && $.active === 0) {
                        quietTime += 50;
                        if (quietTime >= 600) break; // 600ms ciszy = wszystko załadowane
                    } else {
                        quietTime = 0;
                    }
                }

                const targetReg = userJSON.nrRejestracyjny;
                const normalize = s => (s || "").trim().replace(/\s/g, '').toLowerCase();

                const trySelect = async () => {
                    // 1) Spróbuj klasycznie (jeśli <option> już są w DOM)
                    let ok = setSelect2ByText('#reg_number_select', targetReg, true);
                    if (!ok) ok = setSelect2ByText('#reg_number_select', targetReg, false);
                    // 2) Fallback - select2 AJAX
                    if (!ok) ok = await setSelect2Ajax('#reg_number_select', targetReg);
                    return ok;
                };

                const isSelected = () => {
                    try {
                        const data = $('#reg_number_select').select2('data');
                        if (!data) return false;
                        const text = (data.text || data.id || "");
                        return normalize(text).includes(normalize(targetReg));
                    } catch(e) { return false; }
                };

                // Max 5 prób - jeśli pole zostanie zresetowane przez kolejny AJAX, wybierz ponownie
                for (let attempt = 0; attempt < 5; attempt++) {
                    await trySelect();
                    // Czekaj i sprawdź czy wybór "przetrwał"
                    await new Promise(r => setTimeout(r, 600));
                    if (isSelected()) break;
                    // Wybór zniknął - poczekaj chwilę dłużej i spróbuj znowu
                    await new Promise(r => setTimeout(r, 400));
                }
            }
        }

        function handleVehicle(userJSON, protocolType) {
            // Bazujemy na tym co faktycznie jest w protokole, nie na JSONie
            const actualTypeId = parseInt($("#type_id").select2('val')) || 0;
            const actualIsUpgrade = actualTypeId === 2;

            if (!actualIsUpgrade && userJSON.nrRejestracyjny) {
                const currentRej = $("#nr_rejestracyjny").val() || "";
                const currentRejLower = currentRej.trim().toLowerCase();

                // Pobierz markę z protokołu (select2) żeby sprawdzić czy nie pojawia się w polu rejestracji
                let markaWProtokole = "";
                try {
                    const isTrailer = $("#vehicle_type_id").select2('val') == 4;
                    const markaData = $(isTrailer ? '#trailer_brand_id' : '#marka_id').select2('data');
                    if (markaData && markaData.text) markaWProtokole = markaData.text.trim().toLowerCase();
                } catch(e) {}

                const rejZawieraMarke = markaWProtokole !== "" &&
                                        markaWProtokole !== "-- wybierz --" &&
                                        !markaWProtokole.includes("wybierz") &&
                                        currentRejLower.includes(markaWProtokole);

                if (currentRej.trim() === "" || currentRej.trim() === "." || currentRejLower === "inny" || currentRejLower === "ciężarówka" || currentRejLower === "ciezarowka" || rejZawieraMarke) {
                    $("#nr_rejestracyjny").val(userJSON.nrRejestracyjny);
                } else if (currentRej.trim().toUpperCase() !== userJSON.nrRejestracyjny.trim().toUpperCase()) {
                    if (confirm(`Nr rejestracyjny w protokole to "${currentRej.trim()}". Nadpisać wartością z JSON: "${userJSON.nrRejestracyjny}"?`)) {
                        $("#nr_rejestracyjny").val(userJSON.nrRejestracyjny);
                    }
                }
            }

            const currentBoczny = $("#nr_boczny_pojazdu").val() || "";
            if (currentBoczny.trim() === ".") $("#nr_boczny_pojazdu").val('');
            if (userJSON.nrBoczny && (currentBoczny.trim() === "" || currentBoczny.trim() === ".")) {
                $("#nr_boczny_pojazdu").val(userJSON.nrBoczny);
            }

            // Przy upgrade nie ruszamy rodzaju/marki/modelu - dane załadują się automatycznie po wyborze pojazdu
            if (!actualIsUpgrade) {
                // Dla Tauron/PGE uzupełniamy tylko jeśli puste, dla pozostałych flot nadpisujemy zawsze
                const flota = userJSON.flota || "";
                const czyUzupelniacTylkoJesliPuste = _m(flota, "VEFVUk9O", "UEdF");

                // Rodzaj pojazdu
                if (!czyUzupelniacTylkoJesliPuste || isSelect2Empty('#vehicle_type_id')) {
                    let vType = 1;
                    if(userJSON.typPojazdu === 'Osobowy') vType = 2;
                    if(userJSON.typPojazdu === 'Maszyna') vType = 3;
                    if(userJSON.typPojazdu === 'Naczepa') vType = 4;
                    $("#vehicle_type_id").select2("val", vType).trigger('change');
                }

                // Marka
                const markaSelector = $("#vehicle_type_id").select2('val') == 4 ? '#trailer_brand_id' : '#marka_id';
                if (!czyUzupelniacTylkoJesliPuste || isSelect2Empty(markaSelector)) {
                    if (userJSON.marka) setSelect2ByText(markaSelector, userJSON.marka);
                }

                // Model
                const currentModel = $('input[name="model"]').val() || "";
                if (!czyUzupelniacTylkoJesliPuste || currentModel.trim() === "" || currentModel.trim() === ".") {
                    $('input[name="model"]').val(userJSON.model && String(userJSON.model).trim() !== "" ? userJSON.model : ".");
                }
            }
        }

        async function handleRejestrator(userJSON) {
            if(!userJSON.id) return;
            ensureState('[name=rej]', true);

            let typRej = userJSON.modelUrzadzenia || (userJSON.id.toLowerCase().startsWith('h') ? "SE5" : "");
            const idNum = parseInt(userJSON.id);

            const co = userJSON.firma || "";
            const flota = userJSON.flota || "";
            let domyslnaBaza = "[A] gps.ze-it.pl";

            if (_m(co, "T1ZFUklT", "QkFCSU5TS0k=", "SU5UUkEgUy5B")) {
                domyslnaBaza = "[A] gps.framelogic.pl";
            } else if (_m(flota, "RVRPTEw=")) {
                domyslnaBaza = "[A] gps.framelogic.pl";
            } else if (_m(flota, "VEFVUk9O", "VEQgUy5B", "VERQ", "VE9L")) {
                domyslnaBaza = "[A] gps-server.tpe.corp";
                setSelect2ByText("#vehicle_type_id", "pojazd spalinowy", false);
            } else if (_m(flota, "UEdF")) {
                domyslnaBaza = "[A] gps.gkpge.pl";
            }

            const setConfigDb = async (baza) => {
                // Opcje są statyczne – próbuj od razu, polling tylko jeśli element jeszcze nie widoczny
                for (let i = 0; i < 20; i++) {
                    if ($('select[name="config_db_id"]').length && setSelect2ByText('select[name="config_db_id"]', baza, false)) return;
                    await new Promise(r => setTimeout(r, 50));
                }
            };

            if (typRej.startsWith("Sk")) {
                setSelect2ByText("#rodzaj_rejestratora_id", "Setivo");
                document.getElementById("database-config").style.display = "inline-block";
                await setConfigDb(domyslnaBaza);
            } else if (typRej === "Albatros" || (idNum > 99999 && idNum < 999999)) {
                setSelect2ByText("#rodzaj_rejestratora_id", "Albatros");
                if (!userJSON.modelUrzadzenia || userJSON.modelUrzadzenia.trim() === "") {
                    typRej = "Albatros 8.5";
                }
            } else if (typRej.toLowerCase().startsWith("fm") || idNum > 999999) {
                setSelect2ByText("#rodzaj_rejestratora_id", "TELTONIKA");
                document.getElementById("database-config").style.display = "inline-block";
                await setConfigDb(domyslnaBaza);
            }

            if(typRej) setSelect2ByText("#typ_rejestratora_id", typRej);
            $('[name=dscr]').first().val(userJSON.id);
            ensureState('[name=rej_c]', true);

            const $rejRow = $('[name=dscr]').closest('tr');
            $rejRow.find('select').each(function() {
                if (!isSelect2Empty(this)) return;

                const hasNowyOption = $(this).find('option').filter(function() {
                    return $(this).text().trim().toLowerCase() === 'nowy';
                }).length > 0;
                if (hasNowyOption) {
                    setSelect2ByText(this, 'Nowy');
                }
            });
        }

        function handleCAN(userJSON) {
            const can = userJSON.canConfig;
            if (!can || String(can.isCan).toLowerCase() !== "true") return;

            ensureState("#can", true);
            ensureState("#can_c", true);

            const flota = userJSON.flota || "";
            const czyPominac1611 = _m(flota, "VEFVUk9O", "VEQgUy5B", "UEdF", "VERQ", "VE9L");

            const check = (val, id) => { if (val && val.toLowerCase().includes("tak")) ensureState(id, true); };
            check(can.canPredkosc, "#spn84_c");
            if (!czyPominac1611) check(can.canPredkosc, "#spn1611_c");
            check(can.canObroty, "#spn190_c");
            check(can.canDystans, "#spn917_c");
            check(can.canPaliwo, "#spn96_c");
            check(can.canZuzycie, "#spn250_c");

            if(can.canStatusy && can.canStatusy.toLowerCase().includes("tak")) {
                ensureState("#spanstatus_c", true);
                ensureState("#tagid_c", true);
            }
            if(can.canWebasto && can.canWebasto.toLowerCase().includes("tak")) {
                ensureState("#webasto_can_c", true);
                const wColor = document.querySelector('select[name="webasto_can_color"]');
                if (wColor) { wColor.value = "#02488f"; wColor.dispatchEvent(new Event('change')); }
            }
        }

        function handleD8AndTacho(userJSON) {
            const tachoModel = userJSON.modelTacho === "Siemens" ? 1 : 2;
            const czyJestWersja = (userJSON.wersjaTacho && userJSON.wersjaTacho.trim() !== "");

            if (userJSON.d8 && userJSON.d8 !== "Brak") {
                if (userJSON.d8 === "TMR") {
                    ensureState("#tmr", true);
                    forceInput("#tmr_status", userJSON.tmrResponse);
                    $("#tmr_model_id").select2('val', tachoModel).trigger('change');
                    forceInput("#tmr_nr_firmware", userJSON.wersjaTacho);
                } else {
                    ensureState("#kabel_d8", true);
                    $("#kabel_d8_producent_id").select2('val', tachoModel).trigger('change');
                    setSelect2ByText("#kabel_d8_podlaczenie_id", userJSON.d8 === "FMB640/FMC650" ? "FMB640" : "Tachoreader");
                    setTimeout(() => clickRightDivBtn("kabel_d8_c"), 250);
                }
            }

            if (czyJestWersja) {
                ensureState("#tachoreader", userJSON.d8 === "Tachoreader");
                ensureState("#fmb640", userJSON.d8 === "FMB640/FMC650");

                if (userJSON.d8 === "Tachoreader") {
                    $("#tachoreader_model_id").select2('val', tachoModel).trigger('change');
                    forceInput("#tachoreader_status", 2);
                    forceInput("#tachoreader_nr_firmware", userJSON.wersjaTacho);
                    setTimeout(() => clickRightDivBtn("tachoreader_c"), 350);
                } else if (userJSON.d8 === "FMB640/FMC650") {
                    $("#fmb640_model_id").select2('val', tachoModel).trigger('change');
                    forceInput("#fmb640_status", userJSON.wersjaTacho);
                    forceInput("#fmb640_nr_firmware", ".");
                    setTimeout(() => clickRightDivBtn("fmb640_c"), 350);
                }
            } else {
                ensureState("#tachoreader", false);
                ensureState("#fmb640", false);
            }
        }

        async function handleDINs(userJSON, delayMs) {
            if (!userJSON.dinConfig || typeof userJSON.dinConfig !== 'object') return;
            const colors = ["Czerwony", "Żółty", "Khaki", "Srebrny"];
            let cIdx = 0;

            for (const key of Object.keys(userJSON.dinConfig)) {
                const data = userJSON.dinConfig[key];
                let nazwa = data.nazwa || key;
                if (!nazwa) continue;

                const dinNum = parseInt(data.bit);
                const hasBit = !isNaN(dinNum) && String(data.bit).trim() !== "";
                const stan = data.stan ? (data.stan.charAt(0).toUpperCase() + data.stan.slice(1)) : 'Niski';
                const sn = data.sn || "";
                let kolor = "Czarny";

                if (nazwa.toLowerCase() === "webasto") {
                    nazwa = "Webasto"; kolor = "Granatowy";
                } else {
                    if (cIdx < colors.length) kolor = colors[cIdx];
                    cIdx++;
                    nazwa = nazwa.charAt(0).toUpperCase() + nazwa.slice(1);
                }
                await addUrzadzenieDodatkoweDin(nazwa, dinNum, stan, kolor, sn, hasBit);
            }
        }

        async function handleSondy(userJSON) {
            const cleanFloat = (val) => val ? String(val).replace(/\.0$/, '') : '';

            async function buildSonda(num, poj, skal, dataNum, nrWejscia, isRight, prawyZbiornikVal) {
                const cleanPoj = cleanFloat(poj);
                const cleanSkal = cleanFloat(skal);

                if (!num && !cleanPoj) return;

                // Upewnij się że wszystkie poprzednie zbiorniki istnieją w DOM
                for (let n = 1; n <= dataNum; n++) {
                    if (!$(`tr[data-number=${n}]`).length) {
                        document.querySelector(".tanks.plus.fl-tipsy-bottom-right").click();
                        for (let i = 0; i < 100; i++) {
                            if ($(`tr[data-number=${n}]`).length) break;
                            await new Promise(r => setTimeout(r, 20));
                        }
                    }
                }

                const wiersz = document.querySelector(`tr[data-number='${dataNum}']`);

                // Jeśli wiersz jest już zatwierdzony (zielony) – nie ruszamy go
                const goodBtn = wiersz.querySelector('div[data-name="tanks_good[]"]');
                if (goodBtn && goodBtn.classList.contains('checkbox_checked')) return;

                if (wiersz.querySelector('input[name="pojemnosc[]"]')) {
                    wiersz.querySelector('input[name="pojemnosc[]"]').value = cleanPoj || '';
                }

                if (num) {
                    const sondaRadio = wiersz.querySelector('input[type="radio"][value="1"]');
                    if (sondaRadio) {
                        sondaRadio.click();
                        if(sondaRadio.parentElement.classList.contains('TfRadio')) sondaRadio.parentElement.click();
                    }

                    wiersz.children[2].children[1].children[2].click();

                    for (let i = 0; i < 50; i++) {
                        if (wiersz.querySelector('input[name="dlugosc_sondy[]"]')) break;
                        await new Promise(r => setTimeout(r, 20));
                    }

                    const inSkal = wiersz.querySelector('input[name="dlugosc_sondy[]"]');
                    const inNum = wiersz.querySelector('input[name="numer_fabryczny_sondy[]"]');
                    if (inSkal && inNum) {
                        inSkal.value = cleanSkal || '';
                        inNum.value = num;
                        $(wiersz.querySelector('select[name="producent_sondy[]"]')).select2('val', 1);
                        $(wiersz.querySelector('select[name="nr_wejscia[]"]')).select2('val', nrWejscia);

        async function setSelect2Ajax(selector, text) {
            if (!text || text.trim() === "") return false;
            const $el = $(selector);
            if ($el.length === 0) return false;

            const normalize = (s) => s.trim().replace(/\s/g, '').toLowerCase();
            const searchText = normalize(text);

            // Pobierz ID kontenera drop dla tego konkretnego elementu
            const getDropContainer = () => {
                // select2 v3 – kontener ma id bazujący na id selecta
                const elId = $el.attr('id') || $el.closest('.select2-container').attr('id');
                if (elId) {
                    const dropId = '#select2-drop'; // v3 używa jednego globalnego
                }
                return $('#select2-drop');
            };

            const isDropdownOpen = () => {
                const $drop = getDropContainer();
                return $drop.length && $drop.is(':visible') && !$drop.hasClass('select2-display-none');
            };

            const openAndType = async () => {
                // Upewnij się że żaden inny dropdown nie jest otwarty
                try { $('.select2-container.select2-dropdown-open').not($el.next()).find('a').trigger('click'); } catch(e) {}
                await new Promise(r => setTimeout(r, 100));

                try { $el.select2('open'); } catch(e) { return false; }

                // Czekaj aż dropdown będzie widoczny (max 2s)
                for (let i = 0; i < 40; i++) {
                    await new Promise(r => setTimeout(r, 50));
                    if (isDropdownOpen()) break;
                }

                if (!isDropdownOpen()) return false;

                // Znajdź pole input w dropdownie
                const $search = $('#select2-drop input.select2-input, #select2-drop .select2-search input');
                if ($search.length === 0) return false;

                // Wyczyść i wpisz tekst
                $search.focus().val('').trigger('input');
                await new Promise(r => setTimeout(r, 50));
                $search.val(text).trigger('input').trigger('keyup').trigger('keypress');

                return true;
            };

            // Max 3 próby otwarcia
            let typed = false;
            for (let attempt = 0; attempt < 3; attempt++) {
                typed = await openAndType();
                if (typed) break;
                await new Promise(r => setTimeout(r, 400));
            }

            if (!typed) return false;

            let found = false;

            for (let i = 0; i < 30; i++) {
                await new Promise(r => setTimeout(r, 200));

                if (!isDropdownOpen()) {
                    // Dropdown zamknął się – sprawdź czy może już wybrał wartość automatycznie
                    const selected = $el.select2('data');
                    if (selected && selected.text && normalize(selected.text).includes(searchText)) {
                        found = true;
                        break;
                    }
                    // Spróbuj ponownie otworzyć
                    const reopened = await openAndType();
                    if (!reopened) break;
                    continue;
                }

                if ($('#select2-drop .select2-no-results').is(':visible')) break;

                const $results = $('#select2-drop .select2-result-label');
                if ($results.length === 0) continue;

                let exactMatch = null;
                let partialMatch = null;

                $results.each(function() {
                    const resultText = normalize($(this).text());
                    if (resultText === searchText) {
                        exactMatch = $(this);
                        return false;
                    }
                    if (!partialMatch && resultText.includes(searchText)) {
                        partialMatch = $(this);
                    }
                });

                const matched = exactMatch || partialMatch;

                if (matched) {
                    let target = matched.closest('.select2-result');
                    if (target.length === 0) target = matched;

                    const node = target[0];
                    node.dispatchEvent(new MouseEvent('mousedown', { bubbles: true, cancelable: true }));
                    node.dispatchEvent(new MouseEvent('mouseup', { bubbles: true, cancelable: true }));

                    found = true;
                    break;
                }
            }

            if (!found) {
                try { $el.select2('close'); } catch(e) {}
            }
            return found;
        }

        async function trySetSelect2(selector, text, exact = false, isLicensePlate = false, maxAttempts = 15, delayMs = 100) {
            if (!text) return false;
            for (let i = 0; i < maxAttempts; i++) {
                if (setSelect2ByText(selector, text, exact, isLicensePlate)) {
                    return true;
                }
                await new Promise(r => setTimeout(r, delayMs));
            }
            return false;
        }

        function setSelect2ByText(selector, text, exact = true, isLicensePlate = false) {
            if (!text) return false;
            let foundValue = null;
            let $foundSelect = null;

            const normalize = (str) => {
                let s = str.trim().toLowerCase();
                if (!exact) s = s.replace(/\s/g, '').replace(/-/g, '');
                if (isLicensePlate) s = s.replace(/o/g, '0');
                return s;
            };

            const search = normalize(text);

            $(selector).each(function() {
                const $currentSelect = $(this);
                $currentSelect.find('option').each(function() {
                    const optTextRaw = $(this).text();
                    const optText = normalize(optTextRaw);

                    if (exact ? optText === search : optText.includes(search)) {
                        foundValue = $(this).val();
                        $foundSelect = $currentSelect;
                        return false;
                    }
                });
                if (foundValue) return false;
            });

            if (foundValue && $foundSelect) {
                $foundSelect.select2('val', foundValue).trigger('change');
                return true;
            }
            return false;
        }

        function forceInput(selector, value) {

        }

                // Ale potem kod używa newDinTr.children[X] — musimy mieć całe <tr>
                // Wyciągamy <tr> będący rodzicem td.label
                newDinTr = $(`#${newDeviceId}`)[0].closest('tr');
            }

            // Używamy find() zamiast kruchej nawigacji przez children[X].children[Y]
            if (hasBit) {
                $(newDinTr).find('select[name="nrwej[]"]').select2('val', din).trigger('change');

            setSelect2ByText($(newDinTr).find('select[name="color[]"]'), color);

            if (sn !== "") {
                for (let i = 0; i < 30; i++) {
                    if ($(newDinTr).find('input[name="din_content[]"]').length > 0) break;
                    await new Promise(r => setTimeout(r, 50));
                }
                const $snInput = $(newDinTr).find('input[name="din_content[]"]');
                if ($snInput.length > 0) forceInput($snInput, sn);
            }
            if (hasBit) {
                $(newDinTr).find('div[data-name="din_good[]"].checkbox_unchecked').click();
            }
        }
